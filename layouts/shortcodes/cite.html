{{- /* DR pendente (pincite): ogni parametro posizionale è un source id, opzionalmente seguito da "|" per una citazione puntuale (es. "keown1992|111-121"). Sintassi a stringa singola, non un secondo parametro posizionale separato, perché Hugo non permette di mischiare parametri posizionali e nominali nello stesso shortcode, e un secondo posizionale nudo collinderebbe con la sintassi multi-fonte già esistente (es. {{< cite "purser2019" "purser2018" >}}, che unisce due citazioni distinte con "; " — invariata da questa estensione). */ -}} {{- range $i, $rawID := .Params -}} {{- if $i }}; {{ end -}} {{- $parts := split $rawID "|" -}} {{- $sourceID := index $parts 0 -}} {{- $page := "" -}} {{- if gt (len $parts) 1 -}} {{- $page = index $parts 1 -}} {{- end -}} {{- $found := false -}} {{- range hugo.Data.sources -}} {{- if eq .id $sourceID -}} {{- $found = true -}} {{- $authors := .authors -}} {{- $firstAuthorID := index $authors 0 -}} {{- $firstAuthor := "" -}} {{- range hugo.Data.authors -}} {{- if eq .id $firstAuthorID -}} {{- $firstAuthor = .name -}} {{- end -}} {{- end -}} {{- $firstAuthor = replaceRE " [A-Z]\\.?$" "" $firstAuthor -}} {{- if $i }}{{ end -}} {{ $firstAuthor }}{{ if gt (len $authors) 1 }} et al.{{ end }}, {{ .year }}{{ with $page }}: {{ . }}{{ end }} {{- end -}} {{- end -}} {{- if not $found -}} [{{ i18n "source_not_found" }}: {{ $sourceID }}] {{- end -}} {{- end -}}
